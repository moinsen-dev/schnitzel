<project_specification>
  <project_name>Schnitzel Module 02 - Real-time &amp; Events</project_name>
  <module_id>module_02_realtime_events</module_id>
  <depends_on>module_01_api_layer</depends_on>

  <overview>
    Add real-time communication capabilities to the Schnitzel code generator framework:
    event publishing via WebSocket and Redis Pub/Sub, SSE streaming, Temporal job
    scheduling, and BLoC state management generation for Flutter.

    This module creates 6 new generators and 8 new templates that extend the existing
    schnitzel-cli tool built in Module 01.
  </overview>

  <core_features>
    <event_publisher_generator>
      - Generate event publishing infrastructure from schema events section
      - Redis Pub/Sub integration for backend-to-backend communication
      - WebSocket broadcast for client updates
      - Event type definitions with Pydantic models
      - Payload serialization with JSON
      - Channel routing based on event configuration
      - Output: backend/app/generated/events.py
    </event_publisher_generator>

    <sse_stream_generator>
      - Generate SSE endpoints from schema streams section (type: sse)
      - FastAPI StreamingResponse with async generators
      - Chunk type definitions for structured data
      - Reconnection support headers (Last-Event-ID)
      - Event ID tracking for resume capability
      - Output: backend/app/generated/streams.py
    </sse_stream_generator>

    <websocket_handler_generator>
      - Generate WebSocket endpoints from schema streams section (type: websocket)
      - Connection manager for tracking multiple clients
      - Message routing by message type field
      - Heartbeat/keepalive mechanism
      - Authentication via query params or first message
      - Room/channel subscription support
      - Output: backend/app/generated/websocket.py
    </websocket_handler_generator>

    <dart_event_client_generator>
      - Generate event subscription manager for Flutter
      - SSE client with automatic reconnection logic
      - WebSocket client with auto-reconnect and backoff
      - Type-safe event deserialization using freezed models
      - Stream controllers for BLoC integration
      - Connection state management
      - Output: packages/shared/lib/generated/events.dart
    </dart_event_client_generator>

    <temporal_job_generator>
      - Generate Temporal workflow definitions from schema jobs section
      - Activity implementations as method stubs
      - Schedule configuration from cron expressions
      - Retry policies from job configuration
      - Timeout handling with configurable values
      - Output: backend/app/generated/jobs.py
    </temporal_job_generator>

    <bloc_state_generator>
      - Generate feature-specific BLoCs from schema models
      - Event classes: Load, Create, Update, Delete per model
      - State classes: Initial, Loading, Loaded, Error
      - Repository integration pattern for data access
      - Error handling with typed failure classes
      - Optimistic updates support
      - Output: packages/[feature]/lib/bloc/ directory per feature
    </bloc_state_generator>
  </core_features>

  <schema_extensions>
    <events_section>
      Events are defined in the schema YAML under the 'events' key:
      ```yaml
      events:
        order.placed:
          payload:
            order_id: uuid
            customer_id: uuid
            total: decimal
          channels:
            - websocket
            - redis

        inventory.low:
          payload:
            product_id: uuid
            current_stock: integer
            threshold: integer
          channels:
            - redis
      ```
    </events_section>

    <streams_section>
      Streams are defined under the 'streams' key:
      ```yaml
      streams:
        order_updates:
          type: sse
          path: /orders/{order_id}/stream
          events:
            - order.status_changed
            - order.item_added
          auth: required

        chat:
          type: websocket
          path: /ws/chat/{room_id}
          messages:
            - type: message
              payload: {content: string, sender_id: uuid}
            - type: typing
              payload: {user_id: uuid, is_typing: boolean}
          auth: required
      ```
    </streams_section>

    <jobs_section>
      Jobs are defined under the 'jobs' key:
      ```yaml
      jobs:
        daily_report:
          schedule: "0 0 * * *"
          workflow: GenerateDailyReport
          timeout: 30m
          retry_policy:
            max_attempts: 3
            initial_interval: 1m

        cleanup_expired:
          schedule: "*/15 * * * *"
          workflow: CleanupExpiredSessions
          timeout: 5m
      ```
    </jobs_section>
  </schema_extensions>

  <naming_conventions>
    <python>
      - Classes: PascalCase (EventPublisher, ConnectionManager)
      - Functions: snake_case (publish_event, broadcast_message)
      - Files: snake_case (events.py, websocket.py)
      - Constants: UPPER_SNAKE_CASE (DEFAULT_TIMEOUT, MAX_RETRIES)
    </python>
    <dart>
      - Classes: PascalCase (OrderBloc, OrderEvent, OrderState)
      - Files: snake_case (order_bloc.dart, order_event.dart)
      - Variables: camelCase (orderRepository, currentState)
    </dart>
    <events>
      - Event names: dot.notation (order.placed, inventory.low)
      - Channels: lowercase (websocket, redis, sse)
    </events>
    <blocs>
      - BLoC class: {Model}Bloc (OrderBloc, UserBloc)
      - Events: {Model}{Action} (OrderLoad, OrderCreate, OrderUpdate, OrderDelete)
      - States: {Model}State with sealed classes (OrderInitial, OrderLoading, OrderLoaded, OrderError)
    </blocs>
  </naming_conventions>
