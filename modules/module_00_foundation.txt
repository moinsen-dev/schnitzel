  <overview>
    Build the core foundation of the Schnitzel CLI: schema parsing, validation,
    and basic code generation for both Python (Pydantic) and Dart (Freezed) models.
    This phase establishes the generation pipeline from YAML to working code.
  </overview>

  <core_features>
    <schema_parser>
      - Parse YAML schema files with PyYAML
      - Pydantic models for schema validation (SchnitzelSchema, Model, Field, Endpoint, etc.)
      - Multi-file import resolution (imports: section)
      - Feature schema merging
      - Reference validation between models
    </schema_parser>

    <schema_validator>
      - Type consistency checking
      - Relationship validation (belongsTo, hasMany references exist)
      - Naming convention validation (PascalCase models, snake_case fields)
      - Circular dependency detection
      - Required field validation
    </schema_validator>

    <python_model_generator>
      - Generate Pydantic v2 models from schema
      - Field type mapping: string→str, uuid→UUID, int→int, float→float, bool→bool, datetime→datetime, json→dict
      - Validation rules: min, max, format, unique, optional, default
      - Relationships: generate type hints for belongsTo/hasMany
      - JSON serialization with model_dump()
      - Generate models.py in backend/app/generated/
    </python_model_generator>

    <dart_model_generator>
      - Generate Freezed immutable classes from schema
      - Field type mapping: string→String, uuid→String, int→int, float→double, bool→bool, datetime→DateTime, json→Map<String, dynamic>
      - Optional fields with nullable types (String?)
      - Default values in factory constructors
      - JSON serialization with @JsonSerializable
      - copyWith support via Freezed
      - Generate models.dart in packages/shared/lib/generated/
    </dart_model_generator>

    <init_command>
      - schnitzel init <name> [--template minimal|full|ai-chat]
      - Create project directory structure
      - Generate initial schema.schnitzel.yaml from template
      - Run flutter create for packages/app
      - Run uv init for backend/
      - Generate docker-compose.yaml
      - Generate pubspec.yaml (workspace root)
      - Generate pyproject.toml
    </init_command>

    <generate_command>
      - schnitzel generate [--target all|flutter|python|docker]
      - Parse and validate schema
      - Run appropriate generators based on target
      - --watch mode for development (file watcher)
      - --dry-run preview without writing files
      - Output progress with Rich
    </generate_command>

    <docker_compose_generator>
      - Generate docker-compose.yaml from services config
      - PostgreSQL 16 with pgvector extension
      - Redis 7 with pub/sub features
      - Health checks for all services
      - Volume mounts for persistence
      - Port collision prevention (check available ports)
      - Environment variable substitution
    </docker_compose_generator>
  </core_features>

  <database_schema>
    <note>This module creates the CLI tool itself, not application databases.
    The CLI will later generate database schemas for user projects.</note>
  </database_schema>
