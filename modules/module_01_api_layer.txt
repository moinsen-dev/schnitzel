<module_specification>
  <module_name>API Layer</module_name>

  <overview>
    Extend the Schnitzel CLI with full API generation capabilities. This includes
    generating FastAPI routes from endpoint definitions, Dart API clients for
    type-safe HTTP communication, database migrations with Alembic, and the
    serve/validate/migrate commands.
  </overview>

  <core_features>
    <fastapi_route_generator>
      - Generate FastAPI route stubs from endpoints section
      - Path parameter extraction ({id} → id: UUID)
      - Query parameter handling with defaults
      - Request body validation using generated Pydantic models
      - Response serialization with proper status codes
      - OpenAPI metadata (operation_id from name)
      - Auth decorator integration (auth: required → Depends(get_current_user))
      - Role checking (roles: [admin] → Depends(require_roles(["admin"])))
      - Generate routes.py in backend/app/generated/
    </fastapi_route_generator>

    <sqlalchemy_orm_generator>
      - Generate SQLAlchemy 2.0 ORM models from schema models
      - Table definitions with proper column types
      - Primary keys, foreign keys, indexes
      - Relationship definitions (relationship())
      - Generate orm.py in backend/app/generated/
    </sqlalchemy_orm_generator>

    <dart_api_client_generator>
      - Generate Dio-based HTTP client
      - Type-safe method signatures matching endpoints
      - Path parameter interpolation
      - Query parameter encoding
      - Request body serialization
      - Response deserialization to Freezed models
      - Error handling with typed exceptions
      - Auth token injection via interceptor
      - Generate api_client.dart in packages/shared/lib/generated/
    </dart_api_client_generator>

    <database_migration_generator>
      - Alembic integration for migrations
      - Schema diff detection (compare current DB to schema)
      - SQL generation for CREATE TABLE, ALTER TABLE
      - Index creation
      - Foreign key constraints
      - Generate migrations in .schnitzel/migrations/
    </database_migration_generator>

    <serve_command>
      - schnitzel serve [--backend-only] [--port 8000]
      - Start Docker services (docker compose up -d)
      - Run FastAPI with uvicorn (hot reload)
      - Optionally start Flutter dev server
      - Proxy configuration for API calls
      - Display running services with Rich
    </serve_command>

    <validate_command>
      - schnitzel validate [--strict] [--breaking]
      - Run schema validation
      - Check for breaking changes vs last snapshot
      - Naming convention enforcement
      - Security checks (auth on mutations)
      - Output validation report
    </validate_command>

    <migrate_command>
      - schnitzel migrate diff --name "description"
      - schnitzel migrate up
      - schnitzel migrate down
      - schnitzel migrate status
      - Integrate with Alembic
      - Store migration history
    </migrate_command>
  </core_features>
