<module_specification>
  <module_name>Real-time & Events</module_name>
  <module_id>module_02_realtime_events</module_id>
  <depends_on>module_01_api_layer</depends_on>
  <inherit_from>app_spec_master.txt</inherit_from>

  <overview>
    Add real-time communication capabilities: event publishing via WebSocket and
    Redis Pub/Sub, SSE streaming, Temporal job scheduling, and BLoC state management
    generation for Flutter.
  </overview>

  <core_features>
    <event_publisher_generator>
      - Generate event publishing infrastructure from events section
      - Redis Pub/Sub integration for backend-to-backend
      - WebSocket broadcast for client updates
      - Event type definitions with Pydantic
      - Payload serialization
      - Channel routing based on event config
      - Generate events.py in backend/app/generated/
    </event_publisher_generator>

    <sse_stream_generator>
      - Generate SSE endpoints from streams section (type: sse)
      - FastAPI StreamingResponse with async generators
      - Chunk type definitions
      - Reconnection support headers
      - Generate streams.py in backend/app/generated/
    </sse_stream_generator>

    <websocket_handler_generator>
      - Generate WebSocket endpoints from streams section (type: websocket)
      - Connection manager for multiple clients
      - Message routing by type
      - Heartbeat/keepalive
      - Authentication via query params or first message
      - Generate websocket.py in backend/app/generated/
    </websocket_handler_generator>

    <dart_event_client_generator>
      - Generate event subscription manager for Flutter
      - SSE client with reconnection logic
      - WebSocket client with auto-reconnect
      - Type-safe event deserialization
      - Stream controllers for Bloc integration
      - Generate events.dart in packages/shared/lib/generated/
    </dart_event_client_generator>

    <temporal_job_generator>
      - Generate Temporal workflow definitions from jobs section
      - Activity implementations (stub)
      - Schedule configuration from cron expressions
      - Retry policies from job config
      - Timeout handling
      - Generate jobs.py in backend/app/generated/
    </temporal_job_generator>

    <bloc_state_generator>
      - Generate feature-specific BLoCs from models
      - Event classes (Load, Create, Update, Delete)
      - State classes (Initial, Loading, Loaded, Error)
      - Repository integration pattern
      - Error handling with typed failures
      - Generate bloc/ directory per feature package
    </bloc_state_generator>
  </core_features>
