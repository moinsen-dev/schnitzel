<module_specification>
  <module_name>Auth & Security</module_name>
  <module_id>module_03_auth_security</module_id>
  <depends_on>module_02_realtime_events</depends_on>
  <inherit_from>app_spec_master.txt</inherit_from>

  <overview>
    Implement complete authentication and authorization code generation including
    JWT token management, OAuth provider integration (Google, Apple), role-based
    access control (RBAC), and session management with Redis.
  </overview>

  <core_features>
    <jwt_auth_generator>
      - Generate JWT token utilities from auth section
      - Access token generation with configurable expiry
      - Refresh token generation and rotation
      - Token validation middleware
      - Claims management (user_id, roles, custom claims)
      - Configurable algorithms (HS256, RS256)
      - Generate auth/jwt.py in backend/app/generated/
    </jwt_auth_generator>

    <oauth_integration_generator>
      - Generate OAuth flows from auth.providers
      - Google OAuth 2.0 flow (email_password, google)
      - Apple Sign In flow (apple)
      - Magic link email authentication (magic_link)
      - Callback handlers
      - Token exchange logic
      - User creation/linking
      - Generate auth/oauth.py in backend/app/generated/
    </oauth_integration_generator>

    <rbac_permission_generator>
      - Generate role definitions from roles section
      - Permission checking utilities
      - Wildcard permission support (*)
      - Scoped permissions (read:own, update:assigned)
      - FastAPI dependency for role checking
      - Field-level access control
      - Generate auth/rbac.py in backend/app/generated/
    </rbac_permission_generator>

    <session_management_generator>
      - Redis-based session storage
      - Session creation and validation
      - Multi-device support (session per device)
      - Session invalidation (logout, logout all)
      - Remember me functionality
      - Session expiry based on auth.session config
      - Generate auth/sessions.py in backend/app/generated/
    </session_management_generator>

    <mfa_generator>
      - MFA support based on auth.mfa config
      - TOTP (Time-based One-Time Password) generation
      - Backup codes
      - MFA enrollment flow
      - MFA verification middleware
      - Generate auth/mfa.py in backend/app/generated/
    </mfa_generator>

    <dart_auth_client_generator>
      - Token storage (secure storage)
      - Auto token refresh
      - Auth state management (AuthBloc)
      - OAuth flow handlers (url_launcher)
      - Logout and session management
      - Generate auth.dart in packages/shared/lib/generated/
    </dart_auth_client_generator>
  </core_features>
