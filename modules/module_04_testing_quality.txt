<module_specification>
  <module_name>Testing & Quality</module_name>
  <module_id>module_04_testing_quality</module_id>
  <depends_on>module_03_auth_security</depends_on>
  <inherit_from>app_spec_master.txt</inherit_from>

  <overview>
    Add test infrastructure and quality tooling: factory generators for test data,
    mock generators, contract testing, and CLI commands for testing, seeding, and
    schema linting.
  </overview>

  <core_features>
    <python_factory_generator>
      - Generate factory-boy factories from models
      - Faker integration for realistic data
      - Relationship factories (nested creation)
      - Batch generation support
      - Custom field overrides
      - Generate factories.py in backend/app/generated/
    </python_factory_generator>

    <dart_factory_generator>
      - Generate test factory functions for Freezed models
      - Faker package integration
      - Builder pattern for customization
      - Relationship factories
      - Generate factories.dart in packages/shared/lib/generated/
    </dart_factory_generator>

    <mock_generator>
      - Generate repository mocks (Python: unittest.mock)
      - Generate service mocks
      - Generate API client mocks (Dart: mockito)
      - Configurable responses
      - Generate mocks.py and mocks.dart
    </mock_generator>

    <contract_test_generator>
      - Generate API contract tests
      - Request/response schema validation
      - Breaking change detection
      - Pact-style contracts
      - Generate contract_tests.py
    </contract_test_generator>

    <test_command>
      - schnitzel test [--flutter] [--backend] [--contracts] [--coverage]
      - Run pytest for Python tests
      - Run flutter test for Dart tests
      - Run contract tests
      - Coverage reporting
      - Watch mode (--watch)
    </test_command>

    <seed_command>
      - schnitzel seed [--env dev|test]
      - Use factories to generate test data
      - Respect relationships (create dependencies first)
      - Clear existing data option
      - Configurable seed scenarios
    </seed_command>

    <lint_command>
      - schnitzel lint [--fix] [--format json] [--rules strict]
      - Naming convention checks
      - Security rule enforcement
      - Best practices validation
      - Auto-fix capability for safe issues
      - SARIF output for CI integration
    </lint_command>
  </core_features>
