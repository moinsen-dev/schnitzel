<module_specification>
  <module_name>Documentation & AI</module_name>

  <overview>
    Add comprehensive documentation generation (OpenAPI, Swagger UI, ReDoc) and
    AI-first tooling (CLAUDE.md generator, MCP server, AI context files, and
    task-specific prompts).
  </overview>

  <core_features>
    <openapi_generator>
      - Generate OpenAPI 3.1 specification from schema
      - Complete endpoint documentation
      - Request/response schema definitions
      - Authentication documentation
      - Server configuration from environments
      - Export to docs/openapi.yaml
    </openapi_generator>

    <documentation_ui>
      - Configure Swagger UI at /docs
      - Configure ReDoc at /redoc
      - Custom branding from schema meta
      - Authentication UI integration
    </documentation_ui>

    <postman_collection_generator>
      - Generate Postman collection from endpoints
      - Environment variables for base URL
      - Auth token handling
      - Example requests
      - Export to docs/postman.json
    </postman_collection_generator>

    <markdown_docs_generator>
      - Generate static API.md documentation
      - Model documentation with fields
      - Endpoint documentation with examples
      - Event documentation
      - Export to docs/API.md
    </markdown_docs_generator>

    <docs_command>
      - schnitzel docs [--format openapi|postman|markdown|all]
      - schnitzel docs --serve (local doc server)
      - Generate specified format(s)
      - Output to docs/ directory
    </docs_command>

    <claude_md_generator>
      - Generate CLAUDE.md project context file
      - Project overview from meta
      - Model summaries with key fields
      - Endpoint list organized by feature
      - Event and stream documentation
      - Development commands reference
      - Architecture overview
      - Export to project root CLAUDE.md
    </claude_md_generator>

    <mcp_server_generator>
      - Generate MCP (Model Context Protocol) server
      - Schema query tools (list models, get model details)
      - Code generation tools (generate model, generate endpoint)
      - Validation tools (validate schema)
      - Documentation tools (get endpoint docs)
      - Export to .schnitzel/mcp-server/
    </mcp_server_generator>

    <ai_context_generator>
      - Generate structured ai-context.json
      - Machine-readable schema summary
      - Feature dependency graph
      - API endpoint map
      - Export to .schnitzel/ai-context.json
    </ai_context_generator>

    <prompt_generator>
      - Generate task-specific AI prompts
      - add-feature.md - Adding new features
      - fix-bug.md - Bug investigation
      - add-endpoint.md - Adding endpoints
      - add-model.md - Adding models
      - Export to .schnitzel/prompts/
    </prompt_generator>

    <ai_command>
      - schnitzel ai context - Regenerate CLAUDE.md and ai-context.json
      - schnitzel ai mcp - Generate/update MCP server
      - schnitzel ai prompt - Generate task prompts
    </ai_command>
  </core_features>
