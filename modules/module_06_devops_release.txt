<module_specification>
  <module_name>DevOps & Release</module_name>
  <module_id>module_06_devops_release</module_id>

  <inherit_from>app_spec_master.txt</inherit_from>

  <overview>
    Complete the Schnitzel CLI with production-ready DevOps tooling: CI/CD pipeline
    generation, release management with semantic versioning, deployment commands,
    app store metadata generation, and internationalization support.
  </overview>

  <core_features>
    <cicd_pipeline_generator>
      - Generate GitHub Actions workflows from cicd section
      - PR pipeline (lint, validate, test, build)
      - Main branch pipeline (test, build, deploy staging)
      - Release pipeline (test, build, deploy production)
      - Cache configuration (Flutter, Python, Docker)
      - Export to .github/workflows/
    </cicd_pipeline_generator>

    <release_command>
      - schnitzel release --bump patch|minor|major
      - schnitzel release --feature <name> --bump patch|minor|major
      - schnitzel release --dry-run
      - Semantic version bumping
      - Changelog generation (Conventional Commits)
      - Git tagging
      - Update schema version
      - Update pubspec.yaml version
    </release_command>

    <deploy_command>
      - schnitzel deploy --env staging|production
      - schnitzel deploy --store ios|android --track testflight|internal|production
      - Environment-specific deployment
      - Fastlane integration for app stores
      - Backend deployment hooks
      - Rollback support
    </deploy_command>

    <app_store_metadata_generator>
      - Generate iOS App Store Connect metadata
      - Generate Android Play Store metadata
      - Screenshot configuration
      - Release notes from changelog
      - Export to ios/fastlane/metadata and android/fastlane/metadata
    </app_store_metadata_generator>

    <i18n_generator>
      - Generate translation structure for shard_i18n
      - Feature-sharded JSON files
      - assets/i18n/<locale>/<feature>.json
      - Default locale from schema
      - Supported locales configuration
    </i18n_generator>

    <i18n_command>
      - schnitzel i18n extract - Extract translatable strings
      - schnitzel i18n fill --provider deepl|openai - Auto-translate missing
      - schnitzel i18n validate - Check translation coverage
      - Integration with DeepL/OpenAI for auto-translation
    </i18n_command>

    <environment_config_generator>
      - Generate Flutter flavor configuration
      - Generate Gradle build variants
      - Generate XCode schemes
      - Environment-specific bundle IDs
      - Environment-specific API URLs
    </environment_config_generator>

    <add_command>
      - schnitzel add model <name>
      - schnitzel add endpoint <path>
      - schnitzel add feature <name>
      - Interactive schema modification
      - Regenerate affected files
    </add_command>
  </core_features>
