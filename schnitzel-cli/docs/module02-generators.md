## Event Publisher Generator

The Event Publisher generator creates type-safe event publishing infrastructure with support for multiple channels (WebSocket, Redis Pub/Sub, Push notifications, etc.).

### Schema YAML

```yaml
schnitzel: "1.0"

models:
  Order:
    description: "Order in the system"
    fields:
      id:
        type: uuid
        primary: true
      user_id:
        type: uuid
      total:
        type: decimal
      status:
        type: enum
        values: ["pending", "confirmed", "delivered"]

events:
  order.placed:
    description: "Event triggered when an order is placed"
    payload:
      order_id:
        type: uuid
      user_id:
        type: uuid
      total:
        type: decimal
      items_count:
        type: int
      placed_at:
        type: datetime
    channels:
      - websocket
      - redis-pubsub

  order.confirmed:
    description: "Event triggered when an order is confirmed"
    payload:
      order_id:
        type: uuid
      confirmed_at:
        type: datetime
    channels:
      - websocket

  payment.processed:
    description: "Event triggered when payment is processed"
    payload:
      payment_id:
        type: uuid
      order_id:
        type: uuid
      amount:
        type: decimal
      currency:
        type: string
      success:
        type: bool
      metadata:
        type: json
        optional: true
    channels:
      - redis-pubsub
      - push
```

## SSE Stream Generator

Server-Sent Events (SSE) provide unidirectional real-time streaming from server to client.

### Schema YAML

```yaml
schnitzel: "1.0"

streams:
  /orders/{order_id}/status:
    name: order_status_stream
    type: sse
    path: /orders/{order_id}/status
    auth: required
    description: "Real-time order status updates"
    events:
      - order.status_changed
      - order.items_updated
    chunks:
      status:
        type: string
      updated_at:
        type: datetime
      message:
        type: string
        optional: true

  /notifications:
    name: notifications_stream
    type: sse
    path: /notifications
    auth: required
    description: "User notification stream"
    events:
      - notification.received
```

import 'dart:async';
import 'dart:convert';
import 'package:http/http.dart' as http;

class SSEClient {
  final String baseUrl;
  final String? authToken;
  final Duration reconnectDelay;

  http.Client? _httpClient;
  final Map<String, StreamController<Map<String, dynamic>>> _controllers = {};
  final Map<String, bool> _isConnected = {};

  SSEClient({
    required this.baseUrl,
    this.authToken,
    this.reconnectDelay = const Duration(seconds: 3),
  }) {
    _httpClient = http.Client();
  }

  Stream<Map<String, dynamic>> orderStatusStream(String orderId) {
    final streamKey = 'order_status_stream';

    if (_controllers.containsKey(streamKey)) {
      return _controllers[streamKey]!.stream;
    }

    final controller = StreamController<Map<String, dynamic>>.broadcast();
    _controllers[streamKey] = controller;

    Future<void> connect() async {
      if (_isConnected[streamKey] == true) return;
      _isConnected[streamKey] = true;

      try {
        final uri = Uri.parse('$baseUrl/orders/$orderId/status');
        final request = http.Request('GET', uri);

        if (authToken != null) {
          request.headers['Authorization'] = 'Bearer $authToken';
        }
        request.headers['Accept'] = 'text/event-stream';
        request.headers['Cache-Control'] = 'no-cache';

        final response = await _httpClient!.send(request);

        if (response.statusCode != 200) {
          throw Exception('SSE connection failed: ${response.statusCode}');
        }

        response.stream
            .transform(utf8.decoder)
            .transform(const LineSplitter())
            .listen(
              (line) {
                if (line.startsWith('data: ')) {
                  final data = line.substring(6);
                  try {
                    final json = jsonDecode(data) as Map<String, dynamic>;
                    controller.add(json);
                  } catch (e) {
                    // Skip malformed JSON
                  }
                }
              },
              onError: (error) {
                _isConnected[streamKey] = false;
                Future.delayed(reconnectDelay, () => connect());
              },
              onDone: () {
                _isConnected[streamKey] = false;
                Future.delayed(reconnectDelay, () => connect());
              },
              cancelOnError: false,
            );
      } catch (e) {
        _isConnected[streamKey] = false;
        Future.delayed(reconnectDelay, () => connect());
      }
    }

    connect();
    return controller.stream;
  }

  void dispose() {
    for (var controller in _controllers.values) {
      controller.close();
    }
    _controllers.clear();
    _httpClient?.close();
    _httpClient = null;
  }
}

from fastapi import APIRouter, WebSocket, WebSocketDisconnect, Depends
from typing import Dict, Set
import json
import logging

logger = logging.getLogger(__name__)
router = APIRouter()


class ConnectionManager:
    """Manage WebSocket connections."""

    def __init__(self):
        self.active_connections: Dict[str, Set[WebSocket]] = {}

    async def connect(self, websocket: WebSocket, room_id: str):
        await websocket.accept()
        if room_id not in self.active_connections:
            self.active_connections[room_id] = set()
        self.active_connections[room_id].add(websocket)
        logger.info(f"Client connected to room {room_id}")

    def disconnect(self, websocket: WebSocket, room_id: str):
        if room_id in self.active_connections:
            self.active_connections[room_id].discard(websocket)
        logger.info(f"Client disconnected from room {room_id}")

    async def broadcast(self, room_id: str, message: dict):
        """Broadcast message to all connections in room."""
        if room_id in self.active_connections:
            disconnected = set()
            for connection in self.active_connections[room_id]:
                try:
                    await connection.send_json(message)
                except Exception:
                    disconnected.add(connection)

            # Clean up disconnected clients
            for conn in disconnected:
                self.disconnect(conn, room_id)


manager = ConnectionManager()


@router.websocket("/chat/{room_id}")
async def chat_stream(
    websocket: WebSocket,
    room_id: str,
    token: str = None  # Auth via query parameter
):
    """WebSocket endpoint for real-time chat.

    Real-time chat communication
    Auth: required
    """

    # Authenticate user
    user = await authenticate_ws_user(token)
    if not user:
        await websocket.close(code=1008)  # Policy violation
        return

    await manager.connect(websocket, room_id)

    try:
        while True:
            # Receive message
            data = await websocket.receive_json()
            message_type = data.get("type")

            if message_type == "chat.message":
                # Broadcast message to room
                await manager.broadcast(room_id, {
                    "type": "chat.message",
                    "payload": {
                        "user_id": str(user.id),
                        "message": data["payload"]["message"],
                        "timestamp": datetime.utcnow().isoformat()
                    }
                })

            elif message_type == "chat.typing":
                # Broadcast typing indicator
                await manager.broadcast(room_id, {
                    "type": "chat.typing",
                    "payload": {
                        "user_id": str(user.id),
                        "is_typing": data["payload"]["is_typing"]
                    }
                })

    except WebSocketDisconnect:
        manager.disconnect(websocket, room_id)
    except Exception as e:
        logger.error(f"WebSocket error: {e}")
        manager.disconnect(websocket, room_id)

import 'dart:async';
import 'dart:convert';
import 'package:web_socket_channel/web_socket_channel.dart';
import 'package:web_socket_channel/io.dart';

class WebSocketClient {
  final String baseUrl;
  final String? authToken;
  final Duration reconnectDelay;
  final Duration heartbeatInterval;

  final Map<String, WebSocketChannel?> _channels = {};
  final Map<String, StreamController<Map<String, dynamic>>> _controllers = {};
  final Map<String, Timer?> _heartbeatTimers = {};
  final Map<String, bool> _isConnected = {};
  final Map<String, int> _reconnectAttempts = {};

  WebSocketClient({
    required this.baseUrl,
    this.authToken,
    this.reconnectDelay = const Duration(seconds: 3),
    this.heartbeatInterval = const Duration(seconds: 30),
  });

  Stream<Map<String, dynamic>> chatStream(String roomId) {
    final streamKey = 'chat_stream';

    if (_controllers.containsKey(streamKey)) {
      return _controllers[streamKey]!.stream;
    }

    final controller = StreamController<Map<String, dynamic>>.broadcast();
    _controllers[streamKey] = controller;

    Future<void> connect() async {
      if (_isConnected[streamKey] == true) return;

      try {
        final wsUrl = baseUrl.replaceFirst('http://', 'ws://').replaceFirst('https://', 'wss://');
        var uri = Uri.parse('$wsUrl/chat/$roomId');

        if (authToken != null) {
          uri = uri.replace(queryParameters: {
            ...uri.queryParameters,
            'token': authToken!,
          });
        }

        final channel = IOWebSocketChannel.connect(uri);
        _channels[streamKey] = channel;
        _isConnected[streamKey] = true;
        _reconnectAttempts[streamKey] = 0;

        // Start heartbeat
        _heartbeatTimers[streamKey]?.cancel();
        _heartbeatTimers[streamKey] = Timer.periodic(heartbeatInterval, (timer) {
          if (_isConnected[streamKey] == true) {
            channel.sink.add(jsonEncode({'type': 'ping'}));
          }
        });

        channel.stream.listen(
          (message) {
            try {
              final json = jsonDecode(message) as Map<String, dynamic>;
              controller.add(json);
            } catch (e) {
              // Skip malformed JSON
            }
          },
          onError: (error) {
            _isConnected[streamKey] = false;
            _heartbeatTimers[streamKey]?.cancel();
            // Exponential backoff
            final attempts = _reconnectAttempts[streamKey] ?? 0;
            final delay = reconnectDelay * (1 << attempts.clamp(0, 5));
            _reconnectAttempts[streamKey] = attempts + 1;
            Future.delayed(delay, () => connect());
          },
          onDone: () {
            _isConnected[streamKey] = false;
            _heartbeatTimers[streamKey]?.cancel();
            final attempts = _reconnectAttempts[streamKey] ?? 0;
            final delay = reconnectDelay * (1 << attempts.clamp(0, 5));
            _reconnectAttempts[streamKey] = attempts + 1;
            Future.delayed(delay, () => connect());
          },
          cancelOnError: false,
        );
      } catch (e) {
        _isConnected[streamKey] = false;
        final attempts = _reconnectAttempts[streamKey] ?? 0;
        final delay = reconnectDelay * (1 << attempts.clamp(0, 5));
        _reconnectAttempts[streamKey] = attempts + 1;
        Future.delayed(delay, () => connect());
      }
    }

    connect();
    return controller.stream;
  }

  void send(String streamKey, Map<String, dynamic> message) {
    final channel = _channels[streamKey];
    if (channel != null && _isConnected[streamKey] == true) {
      channel.sink.add(jsonEncode(message));
    }
  }

  void dispose() {
    for (var channel in _channels.values) {
      channel?.sink.close();
    }
    _channels.clear();

    for (var controller in _controllers.values) {
      controller.close();
    }
    _controllers.clear();

    for (var timer in _heartbeatTimers.values) {
      timer?.cancel();
    }
    _heartbeatTimers.clear();
  }
}

    authToken: 'your_jwt_token',
  );

    if (message['type'] == 'chat.message') {

    } else if (message['type'] == 'chat.typing') {
      print('User ${message['payload']['user_id']} is typing: ${message['payload']['is_typing']}');
    }
  });

  // Send a message

    'type': 'chat.message',
    'payload': {
      'message': 'Hello, world!'
    }
  });
}

### Generated Dart BLoC Code

```dart
// Generated by Schnitzel Framework
import 'dart:async';
import 'package:bloc/bloc.dart';
import 'package:freezed_annotation/freezed_annotation.dart';
import 'events.dart';  // Generated event client

part 'order_bloc.freezed.dart';

// Events
@freezed
class OrderEvent with _$OrderEvent {
  const factory OrderEvent.started() = _Started;
  const factory OrderEvent.orderUpdated(OrderUpdatedEvent event) = _OrderUpdated;
  const factory OrderEvent.connectionError(String error) = _ConnectionError;
}

// States
@freezed
class OrderState with _$OrderState {
  const factory OrderState.initial() = _Initial;
  const factory OrderState.loading() = _Loading;
  const factory OrderState.loaded({
    required String orderId,
    required String status,
    required DateTime updatedAt,
  }) = _Loaded;
  const factory OrderState.error(String message) = _Error;
}

// BLoC
class OrderBloc extends Bloc<OrderEvent, OrderState> {
  final WebSocketClient _wsClient;
  StreamSubscription? _subscription;

  OrderBloc(this._wsClient) : super(const OrderState.initial()) {
    on<OrderEvent>((event, emit) async {
      await event.when(
        started: () => _onStarted(emit),
        orderUpdated: (orderEvent) => _onOrderUpdated(orderEvent, emit),
        connectionError: (error) => _onConnectionError(error, emit),
      );
    });
  }

  Future<void> _onStarted(Emitter<OrderState> emit) async {
    emit(const OrderState.loading());

    try {
      // Subscribe to WebSocket stream
      _subscription = _wsClient.orderUpdatesStream().listen(
        (data) {
          final event = OrderUpdatedEvent.fromJson(data);
          add(OrderEvent.orderUpdated(event));
        },
        onError: (error) {
          add(OrderEvent.connectionError(error.toString()));
        },
      );
    } catch (e) {
      emit(OrderState.error(e.toString()));
    }
  }

  Future<void> _onOrderUpdated(
    OrderUpdatedEvent event,
    Emitter<OrderState> emit,
  ) async {
    emit(OrderState.loaded(
      orderId: event.orderId,
      status: event.status,
      updatedAt: event.updatedAt,
    ));
  }

  Future<void> _onConnectionError(
    String error,
    Emitter<OrderState> emit,
  ) async {
    emit(OrderState.error('Connection error: $error'));
  }

  @override
  Future<void> close() {
    _subscription?.cancel();
    return super.close();
  }
}
```

## Best Practices

1. **Event Naming**: Use dot.notation (e.g., `order.placed`, `user.registered`) for clear event hierarchies
2. **Channels**: Choose appropriate channels based on delivery requirements:
   - `websocket` for real-time UI updates
   - `redis-pubsub` for service-to-service communication
   - `push` for mobile notifications
3. **Job Scheduling**: Use cron expressions for scheduled jobs, omit schedule for manual-trigger workflows
4. **Parameters**: Define typed parameters for workflows that need input data
5. **Retry Policies**: Configure appropriate retry settings based on job criticality
6. **Streams**: Use SSE for one-way server-to-client, WebSocket for bidirectional communication
