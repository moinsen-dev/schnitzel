  <overview>
    Generate test factories, mocks, and fakes for both Python (factory_boy) and
    Dart (custom factories with Mockito integration). This enables proper testing
    of generated applications.
  </overview>

  <core_features>
    <dart_test_factories>
      - Generate UserFactory.create() pattern for each model
      - Support field overrides: UserFactory.create(email: 'custom@test.com')
      - Generate realistic fake data (names, emails, dates, UUIDs)
      - Integration with faker_dart package
      - Output to: apps/{app}/test/factories/
    </dart_test_factories>

    <python_test_factories>
      - Generate factory_boy factories for each SQLAlchemy model
      - Lazy attributes for related models
      - Sequences for unique fields (email_1, email_2, etc.)
      - Output to: backend/app/tests/factories/
    </python_test_factories>

    <dart_mocks_fakes>
      - Generate MockUserRepository using Mockito
      - Generate FakeUserRepository with in-memory storage
      - Proper type annotations for flutter_test
      - Output to: apps/{app}/test/mocks/
    </dart_mocks_fakes>

    <python_mocks>
      - Generate pytest fixtures for repository mocks
      - Integration with unittest.mock
      - Output to: backend/app/tests/conftest.py additions
    </python_mocks>
  </core_features>

// Generated by Schnitzel Framework
import 'package:faker_dart/faker_dart.dart';
import 'package:my_blog_shared/models/models.dart';

class UserFactory {
  static final _faker = Faker();

  static User create({
    String? id,
    String? email,
    String? name,
    DateTime? createdAt,
  }) {
    return User(
      id: id ?? _faker.datatype.uuid(),
      email: email ?? _faker.internet.email(),
      name: name ?? _faker.person.fullName(),
      createdAt: createdAt ?? DateTime.now(),
    );
  }

  static List<User> createList(int count) {
    return List.generate(count, (_) => create());
  }
}
