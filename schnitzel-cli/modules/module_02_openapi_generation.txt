<module_specification>
  <module_name>OpenAPI Specification Generation</module_name>
  <module_id>module_02_openapi_generation</module_id>
  <depends_on>module_01_field_validation</depends_on>
  <inherit_from>app_spec_master.txt</inherit_from>
  <priority>HIGH</priority>

  <overview>
    Generate OpenAPI 3.0 specification from the Schnitzel schema. This enables API
    documentation, client generation, and contract testing.
  </overview>

  <core_features>
    <openapi_export>
      - Generate docs/openapi.yaml from schema
      - Include all CRUD endpoints for models
      - Include custom endpoints from schema
      - Generate request/response schemas
      - Include authentication requirements
    </openapi_export>

    <docs_command>
      - Add `schnitzel docs` CLI command
      - Options: --format (yaml, json), --output, --serve
      - Serve option starts local Swagger UI
    </docs_command>

    <postman_export>
      - Generate docs/postman_collection.json
      - Include environment variables
      - Group by resource/model
    </postman_export>
  </core_features>

  <files_to_create>
    <generator>src/schnitzel/generators/docs/openapi.py</generator>
    <generator>src/schnitzel/generators/docs/postman.py</generator>
    <command>src/schnitzel/cli/commands/docs.py</command>
  </files_to_create>

  <files_to_modify>
    <file>src/schnitzel/cli.py</file>
    <reason>Register docs command</reason>
  </files_to_modify>

  <implementation_steps>
    <step number="1">
      <title>OpenAPI Generator</title>
      <tasks>
        - Create OpenAPIGenerator class
        - Map Schnitzel types to OpenAPI types
        - Generate paths from CRUD models and custom endpoints
        - Generate components/schemas from models
        - Include securitySchemes for JWT auth
      </tasks>
    </step>

    <step number="2">
      <title>Postman Collection Generator</title>
      <tasks>
        - Create PostmanGenerator class
        - Generate collection with folders per resource
        - Include example request bodies
        - Add environment file with {{baseUrl}}, {{token}}
      </tasks>
    </step>

    <step number="3">
      <title>CLI Command</title>
      <tasks>
        - Create docs.py with Click command
        - Implement --format flag for yaml/json output
        - Implement --serve flag to start Swagger UI
        - Add to schnitzel CLI group
      </tasks>
    </step>

    <step number="4">
      <title>Integration with Generate</title>
      <tasks>
        - Optionally include docs generation in `schnitzel generate`
        - Add --docs flag to generate command
        - Output to docs/ directory
      </tasks>
    </step>
  </implementation_steps>

  <example_output>
    <openapi>
```yaml
openapi: "3.0.3"
info:
  title: My Blog API
  version: "1.0.0"
  description: Generated by Schnitzel Framework

servers:
  - url: http://localhost:8000
    description: Development server

paths:
  /api/users:
    get:
      summary: List users
      operationId: listUsers
      parameters:
        - name: page
          in: query
          schema:
            type: integer
            default: 1
        - name: limit
          in: query
          schema:
            type: integer
            default: 20
      responses:
        '200':
          description: List of users
          content:
            application/json:
              schema:
                type: array
                items:
                  $ref: '#/components/schemas/User'
      security:
        - bearerAuth: []

components:
  schemas:
    User:
      type: object
      required:
        - id
        - email
        - name
      properties:
        id:
          type: string
          format: uuid
        email:
          type: string
          format: email
        name:
          type: string
        created_at:
          type: string
          format: date-time

  securitySchemes:
    bearerAuth:
      type: http
      scheme: bearer
      bearerFormat: JWT
```
    </openapi>
  </example_output>

  <success_criteria>
    - OpenAPI spec validates with swagger-cli validate
    - Postman collection imports without errors
    - schnitzel docs --serve opens working Swagger UI
    - All endpoints documented with request/response schemas
  </success_criteria>
