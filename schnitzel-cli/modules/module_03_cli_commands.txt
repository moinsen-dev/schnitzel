  <overview>
    Implement missing CLI commands for better developer experience:
    - schnitzel lint: Validate schema with best practices
    - schnitzel add: Interactively add models/endpoints
    - schnitzel test: Run Flutter and backend tests
  </overview>

  <core_features>
    <lint_command>
      - Validate schema syntax and structure
      - Check for naming conventions (snake_case models, etc.)
      - Warn about missing indexes on foreign keys
      - Check for circular dependencies
      - Suggest optimizations
      - Output: warnings, errors, suggestions
    </lint_command>

    <add_command>
      - Interactive prompts to add new model
      - Prompts: name, fields (type, optional, default), relations
      - Interactive prompts to add new endpoint
      - Updates schema.schnitzel.yaml automatically
      - Option: --model, --endpoint, --field for non-interactive
    </add_command>

    <test_command>
      - Run Flutter tests: flutter test
      - Run Python tests: pytest
      - Options: --flutter-only, --python-only, --coverage
      - Show unified test results
    </test_command>
  </core_features>
